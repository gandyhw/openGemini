func IsMemUsageExceeded() bool {
    memLimitPct := GetMemUsageLimit()
    if memLimitPct < 1 || memLimitPct >= 100 {
        return false
    }
    memUsedPct := memory.GetMemMonitor().MemUsedPct()
    exceeded := memUsedPct > float64(memLimitPct)
    return exceeded
}
