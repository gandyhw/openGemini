func (rds *RaftDiskStorage) TrySync() error {
    if rds.SyncInterval == 0 || rds.firstSync {
        // ... immediate sync for first write
        return nil
    }
    if !rds.syncTaskCount.CompareAndSwap(0, 1) {
        return nil  // <-- If a sync is already in progress, silently skip!
    }
    go func() {
        rds.backSync()  // <-- Async goroutine
    }()
    return nil
}

func (rds *RaftDiskStorage) backSync() {
    time.Sleep(rds.SyncInterval)  // <-- Sleeps for SyncInterval (default 100ms) before sync
    // ... TrySync on meta and entry files
}

Impact: WAL writes are acknowledged to application before being fsynced.

writes that skipped sync due to syncTaskCount CAS). At 1B logs/s, this is
